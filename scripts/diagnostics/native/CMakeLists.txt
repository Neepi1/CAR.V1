cmake_minimum_required(VERSION 3.16)
project(navlite_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(rosbag2_storage REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(navlite_raw_capture navlite_raw_capture.cpp)
ament_target_dependencies(navlite_raw_capture rclcpp rosbag2_cpp rosbag2_storage)
target_link_libraries(navlite_raw_capture yaml-cpp)
target_compile_options(navlite_raw_capture PRIVATE -Wall -Wextra -Wpedantic)
# Standalone diagnostic build only: no workspace packages or install targets.
option(NAVLITE_BUILD_MPPI_REPLAY "Build the offline native MPPI snapshot integrator" OFF)
set(NAVLITE_NAV_CONFIG_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../src/robot_nav_config" CACHE PATH
  "robot_nav_config source directory (candidate checkout permitted)")
if(NAVLITE_BUILD_MPPI_REPLAY)
  find_package(nav2_mppi_controller REQUIRED)
  find_package(nav2_costmap_2d REQUIRED)
  find_package(nav2_util REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(navlite_mppi_replay "${NAVLITE_NAV_CONFIG_DIR}/tools/navlite_mppi_replay.cpp")
  target_include_directories(navlite_mppi_replay PRIVATE "${NAVLITE_NAV_CONFIG_DIR}/include")
  ament_target_dependencies(navlite_mppi_replay nav2_mppi_controller nav2_costmap_2d
    nav2_util rclcpp nav_msgs geometry_msgs)
  target_link_libraries(navlite_mppi_replay Threads::Threads)
  target_compile_options(navlite_mppi_replay PRIVATE -Wall -Wextra -Wpedantic)
endif()
