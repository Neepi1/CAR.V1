cmake_minimum_required(VERSION 3.16)
project(navlite_snapshot_isolated LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Threads REQUIRED)
foreach(package nav2_mppi_controller nav2_costmap_2d nav2_util nav2_controller nav2_core rclcpp nav_msgs geometry_msgs pluginlib)
  find_package(${package} REQUIRED)
endforeach()
find_library(goal_checker_library simple_goal_checker HINTS /opt/ros/humble/lib REQUIRED)
add_executable(test_transport "${NAV_SOURCE}/test/chassis_dynamics/test_navlite_snapshot.cpp")
target_include_directories(test_transport PRIVATE "${NAV_SOURCE}/include")
target_link_libraries(test_transport Threads::Threads)
add_executable(test_integration
  "${NAV_SOURCE}/test/chassis_dynamics/test_navlite_snapshot_integration.cpp"
  "${NAV_SOURCE}/test/chassis_dynamics/test_navlite_snapshot_visualization_stub.cpp")
add_executable(navlite_mppi_replay "${NAV_SOURCE}/tools/navlite_mppi_replay.cpp")
add_library(navlite_snapshot_candidate SHARED "${NAV_SOURCE}/src/chassis_dynamics/mppi_controller.cpp")
foreach(target test_integration navlite_mppi_replay navlite_snapshot_candidate)
  target_include_directories(${target} PRIVATE "${NAV_SOURCE}/include")
  ament_target_dependencies(${target} nav2_mppi_controller nav2_costmap_2d nav2_util nav2_controller nav2_core rclcpp nav_msgs geometry_msgs pluginlib)
  target_link_libraries(${target} Threads::Threads)
endforeach()
target_link_libraries(test_integration "${goal_checker_library}")
foreach(target test_transport test_integration)
  target_compile_options(${target} PRIVATE -UNDEBUG)
endforeach()
